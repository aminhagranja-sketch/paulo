cmake_minimum_required(VERSION 3.24)
project(MeuGalinheiro VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
option(GRANJA_BUILD_GAME "Build SFML graphical game" ON)
option(GRANJA_BUILD_TESTS "Build gameplay tests" ON)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.12.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(json)
add_library(granja_core src/World.cpp src/Simulation.cpp src/Save.cpp)
target_include_directories(granja_core PUBLIC include)
target_link_libraries(granja_core PUBLIC nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(granja_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(granja_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(GRANJA_BUILD_GAME)
  set(SFML_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
  set(SFML_BUILD_NETWORK OFF CACHE BOOL "" FORCE)
  set(SFML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SFML_BUILD_TEST_SUITE OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(sfml GIT_REPOSITORY https://github.com/SFML/SFML.git GIT_TAG 3.0.2 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(sfml)
  add_executable(meu_galinheiro src/main.cpp src/Game.cpp src/Renderer.cpp)
  target_link_libraries(meu_galinheiro PRIVATE granja_core SFML::Graphics)
  if(MSVC)
    target_compile_options(meu_galinheiro PRIVATE /utf-8 /W4)
  endif()
  add_custom_command(TARGET meu_galinheiro POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:meu_galinheiro>/assets")
  install(TARGETS meu_galinheiro RUNTIME DESTINATION . COMPONENT Runtime)
  install(DIRECTORY assets DESTINATION . COMPONENT Runtime)
  install(FILES README.md LICENSE THIRD_PARTY.md DESTINATION . COMPONENT Runtime)
  install(DIRECTORY docs DESTINATION . COMPONENT Runtime)
endif()
if(GRANJA_BUILD_TESTS)
  enable_testing()
  add_executable(granja_tests tests/core_tests.cpp)
  target_link_libraries(granja_tests PRIVATE granja_core)
  add_test(NAME gameplay COMMAND granja_tests)
endif()
